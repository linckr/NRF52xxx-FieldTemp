# 微信小程序接入温湿度BLE服务方案

## 1. 方案概述

微信小程序支持完整的 BLE (Bluetooth Low Energy) 功能，可以连接 E104-BT5010A 蓝牙温湿度采集设备，访问所有 GATT 服务，包括：

- ✅ **温湿度服务**：读取和订阅温湿度数据
- ✅ **时间同步服务**：向设备发送时间
- ✅ **配置服务**：读取和配置设备参数

## 2. 前置条件

### 2.1 微信小程序配置

在 `app.json` 中配置蓝牙权限：

```json
{
  "permission": {
    "scope.bluetooth": {
      "desc": "需要蓝牙权限连接温湿度设备"
    }
  },
  "requiredPrivateInfos": ["getBluetoothDevices", "onBluetoothDeviceFound"]
}
```

### 2.2 系统权限

- **Android**：需要定位权限（系统将 BLE 扫描与定位关联）
- **iOS**：需要蓝牙权限，且在系统设置中允许小程序访问蓝牙

### 2.3 设备信息

| 项目 | 值 |
|------|-----|
| **设备名称** | `yodfz-temp` |
| **服务 UUID** | `12340001-1234-5678-1234-56789abcdef0` (温湿度服务) |
| **温度特性 UUID** | `12340002-1234-5678-1234-56789abcdef0` |
| **湿度特性 UUID** | `12340003-1234-5678-1234-56789abcdef0` |
| **时间服务 UUID** | `12340010-1234-5678-1234-56789abcdef0` |
| **配置服务 UUID** | `12340020-1234-5678-1234-56789abcdef0` |

**注意**：UUID 必须使用**小写字母**和完整 128 位格式，以确保 Android 和 iOS 平台兼容。

## 3. 完整接入代码示例

### 3.1 设备连接管理类

创建 `utils/bleManager.js`：

```javascript
// 温湿度设备 BLE 管理器
class BLEManager {
  constructor() {
    this.deviceId = null;
    this.serviceId = null;
    this.tempCharId = null;
    this.humidityCharId = null;
    this.isConnected = false;
    this.onTempHumidityChange = null; // 温湿度变化回调
  }

  // 打开蓝牙适配器
  async openAdapter() {
    return new Promise((resolve, reject) => {
      wx.openBluetoothAdapter({
        success: (res) => {
          console.log('蓝牙适配器打开成功', res);
          resolve(res);
        },
        fail: (err) => {
          console.error('蓝牙适配器打开失败', err);
          reject(err);
        }
      });
    });
  }

  // 开始扫描设备
  async startScan() {
    return new Promise((resolve, reject) => {
      wx.startBluetoothDevicesDiscovery({
        services: ['12340001-1234-5678-1234-56789abcdef0'], // 可选：指定服务UUID过滤
        allowDuplicatesKey: false,
        interval: 0,
        success: (res) => {
          console.log('开始扫描设备', res);
          resolve(res);
        },
        fail: (err) => {
          console.error('扫描失败', err);
          reject(err);
        }
      });
    });
  }

  // 监听设备发现
  onDeviceFound(callback) {
    wx.onBluetoothDeviceFound((res) => {
      const devices = res.devices || [];
      devices.forEach(device => {
        // 通过设备名称或服务UUID匹配
        if (device.name === 'yodfz-temp' || 
            (device.advertisServiceUUIDs && 
             device.advertisServiceUUIDs.includes('12340001-1234-5678-1234-56789abcdef0'))) {
          console.log('发现目标设备', device);
          callback(device);
        }
      });
    });
  }

  // 停止扫描
  stopScan() {
    wx.stopBluetoothDevicesDiscovery();
    wx.offBluetoothDeviceFound();
  }

  // 连接设备
  async connectDevice(deviceId) {
    return new Promise((resolve, reject) => {
      wx.createBLEConnection({
        deviceId: deviceId,
        success: (res) => {
          console.log('设备连接成功', res);
          this.deviceId = deviceId;
          this.isConnected = true;
          resolve(res);
        },
        fail: (err) => {
          console.error('设备连接失败', err);
          this.isConnected = false;
          reject(err);
        }
      });
    });
  }

  // 断开连接
  async disconnectDevice() {
    if (!this.deviceId) return;
    
    return new Promise((resolve, reject) => {
      wx.closeBLEConnection({
        deviceId: this.deviceId,
        success: (res) => {
          console.log('设备断开成功', res);
          this.deviceId = null;
          this.isConnected = false;
          resolve(res);
        },
        fail: (err) => {
          console.error('设备断开失败', err);
          reject(err);
        }
      });
    });
  }

  // 获取服务列表
  async getServices() {
    return new Promise((resolve, reject) => {
      wx.getBLEDeviceServices({
        deviceId: this.deviceId,
        success: (res) => {
          console.log('获取服务列表成功', res);
          resolve(res.services || []);
        },
        fail: (err) => {
          console.error('获取服务列表失败', err);
          reject(err);
        }
      });
    });
  }

  // 获取特征值列表
  async getCharacteristics(serviceId) {
    return new Promise((resolve, reject) => {
      wx.getBLEDeviceCharacteristics({
        deviceId: this.deviceId,
        serviceId: serviceId,
        success: (res) => {
          console.log('获取特征值列表成功', res);
          resolve(res.characteristics || []);
        },
        fail: (err) => {
          console.error('获取特征值列表失败', err);
          reject(err);
        }
      });
    });
  }

  // 初始化温湿度服务
  async initTempHumidityService() {
    try {
      // 1. 获取服务列表
      const services = await this.getServices();
      const envService = services.find(s => 
        s.uuid.toLowerCase() === '12340001-1234-5678-1234-56789abcdef0'
      );

      if (!envService) {
        throw new Error('未找到温湿度服务');
      }

      this.serviceId = envService.uuid;
      console.log('找到温湿度服务', this.serviceId);

      // 2. 获取特征值列表
      const characteristics = await this.getCharacteristics(this.serviceId);
      
      // 查找温度特征
      const tempChar = characteristics.find(c => 
        c.uuid.toLowerCase() === '12340002-1234-5678-1234-56789abcdef0' &&
        c.properties.read && c.properties.notify
      );

      // 查找湿度特征
      const humidityChar = characteristics.find(c => 
        c.uuid.toLowerCase() === '12340003-1234-5678-1234-56789abcdef0' &&
        c.properties.read && c.properties.notify
      );

      if (!tempChar || !humidityChar) {
        throw new Error('未找到温湿度特征值');
      }

      this.tempCharId = tempChar.uuid;
      this.humidityCharId = humidityChar.uuid;

      console.log('温度特征', this.tempCharId);
      console.log('湿度特征', this.humidityCharId);

      // 3. 启用通知
      await this.enableNotify(this.tempCharId);
      await this.enableNotify(this.humidityCharId);

      // 4. 监听数据变化
      this.onCharacteristicValueChange();

      // 5. 读取一次当前值
      await this.readTemp();
      await this.readHumidity();

      return true;
    } catch (error) {
      console.error('初始化温湿度服务失败', error);
      throw error;
    }
  }

  // 启用通知
  async enableNotify(characteristicId) {
    return new Promise((resolve, reject) => {
      wx.notifyBLECharacteristicValueChange({
        deviceId: this.deviceId,
        serviceId: this.serviceId,
        characteristicId: characteristicId,
        state: true, // 启用通知
        success: (res) => {
          console.log('启用通知成功', characteristicId);
          resolve(res);
        },
        fail: (err) => {
          console.error('启用通知失败', characteristicId, err);
          reject(err);
        }
      });
    });
  }

  // 监听特征值变化
  onCharacteristicValueChange() {
    wx.onBLECharacteristicValueChange((res) => {
      const { characteristicId, value } = res;
      
      // 解析数据（小端序）
      if (characteristicId.toLowerCase() === this.tempCharId.toLowerCase()) {
        // 温度：int16_t，单位 0.01°C
        const temp = this.readInt16LE(value);
        const tempValue = temp / 100.0;
        
        console.log('温度更新:', tempValue, '°C');
        
        if (this.onTempHumidityChange) {
          this.onTempHumidityChange({
            type: 'temperature',
            value: tempValue,
            raw: temp
          });
        }
      } else if (characteristicId.toLowerCase() === this.humidityCharId.toLowerCase()) {
        // 湿度：uint16_t，单位 0.01%
        const humidity = this.readUInt16LE(value);
        const humidityValue = humidity / 100.0;
        
        console.log('湿度更新:', humidityValue, '%');
        
        if (this.onTempHumidityChange) {
          this.onTempHumidityChange({
            type: 'humidity',
            value: humidityValue,
            raw: humidity
          });
        }
      }
    });
  }

  // 读取温度
  async readTemp() {
    return new Promise((resolve, reject) => {
      wx.readBLECharacteristicValue({
        deviceId: this.deviceId,
        serviceId: this.serviceId,
        characteristicId: this.tempCharId,
        success: (res) => {
          // 数据会在 onBLECharacteristicValueChange 回调中返回
          resolve(res);
        },
        fail: (err) => {
          console.error('读取温度失败', err);
          reject(err);
        }
      });
    });
  }

  // 读取湿度
  async readHumidity() {
    return new Promise((resolve, reject) => {
      wx.readBLECharacteristicValue({
        deviceId: this.deviceId,
        serviceId: this.serviceId,
        characteristicId: this.humidityCharId,
        success: (res) => {
          // 数据会在 onBLECharacteristicValueChange 回调中返回
          resolve(res);
        },
        fail: (err) => {
          console.error('读取湿度失败', err);
          reject(err);
        }
      });
    });
  }

  // 解析 int16_t (小端序)
  readInt16LE(buffer) {
    const view = new DataView(new Uint8Array(buffer).buffer);
    return view.getInt16(0, true); // true 表示小端序
  }

  // 解析 uint16_t (小端序)
  readUInt16LE(buffer) {
    const view = new DataView(new Uint8Array(buffer).buffer);
    return view.getUint16(0, true); // true 表示小端序
  }

  // 监听连接状态
  onConnectionStateChange(callback) {
    wx.onBLEConnectionStateChange((res) => {
      console.log('连接状态变化', res);
      this.isConnected = res.connected;
      if (callback) {
        callback(res);
      }
    });
  }

  // 清理资源
  destroy() {
    if (this.deviceId) {
      this.disconnectDevice();
    }
    wx.offBluetoothDeviceFound();
    wx.offBLECharacteristicValueChange();
    wx.offBLEConnectionStateChange();
    wx.closeBluetoothAdapter();
  }
}

module.exports = BLEManager;
```

### 3.2 页面使用示例

创建 `pages/index/index.js`：

```javascript
const BLEManager = require('../../utils/bleManager.js');

Page({
  data: {
    isConnected: false,
    isScanning: false,
    temperature: null,
    humidity: null,
    deviceList: [],
    selectedDevice: null
  },

  onLoad() {
    this.bleManager = new BLEManager();
    
    // 设置温湿度变化回调
    this.bleManager.onTempHumidityChange = (data) => {
      if (data.type === 'temperature') {
        this.setData({ temperature: data.value.toFixed(2) });
      } else if (data.type === 'humidity') {
        this.setData({ humidity: data.value.toFixed(2) });
      }
    };

    // 监听连接状态
    this.bleManager.onConnectionStateChange((res) => {
      this.setData({ isConnected: res.connected });
      if (!res.connected) {
        wx.showToast({
          title: '设备已断开',
          icon: 'none'
        });
      }
    });
  },

  // 打开蓝牙并扫描
  async startScan() {
    try {
      wx.showLoading({ title: '初始化蓝牙...' });
      
      // 1. 打开蓝牙适配器
      await this.bleManager.openAdapter();
      
      // 2. 监听设备发现
      this.bleManager.onDeviceFound((device) => {
        const devices = this.data.deviceList;
        const exists = devices.find(d => d.deviceId === device.deviceId);
        if (!exists) {
          devices.push({
            deviceId: device.deviceId,
            name: device.name || '未知设备',
            RSSI: device.RSSI
          });
          this.setData({ deviceList: devices });
        }
      });
      
      // 3. 开始扫描
      await this.bleManager.startScan();
      this.setData({ isScanning: true });
      
      wx.hideLoading();
      wx.showToast({
        title: '开始扫描设备',
        icon: 'success'
      });

      // 30秒后自动停止扫描
      setTimeout(() => {
        this.stopScan();
      }, 30000);
      
    } catch (error) {
      wx.hideLoading();
      wx.showModal({
        title: '错误',
        content: error.errMsg || '操作失败',
        showCancel: false
      });
    }
  },

  // 停止扫描
  stopScan() {
    this.bleManager.stopScan();
    this.setData({ isScanning: false });
  },

  // 连接设备
  async connectDevice(e) {
    const deviceId = e.currentTarget.dataset.deviceid;
    
    try {
      wx.showLoading({ title: '连接设备...' });
      
      // 1. 停止扫描
      this.stopScan();
      
      // 2. 连接设备
      await this.bleManager.connectDevice(deviceId);
      
      // 3. 初始化温湿度服务
      await this.bleManager.initTempHumidityService();
      
      wx.hideLoading();
      wx.showToast({
        title: '连接成功',
        icon: 'success'
      });
      
      this.setData({ 
        isConnected: true,
        selectedDevice: deviceId
      });
      
    } catch (error) {
      wx.hideLoading();
      wx.showModal({
        title: '连接失败',
        content: error.errMsg || '无法连接设备',
        showCancel: false
      });
    }
  },

  // 断开连接
  async disconnect() {
    try {
      await this.bleManager.disconnectDevice();
      this.setData({ 
        isConnected: false,
        temperature: null,
        humidity: null
      });
      wx.showToast({
        title: '已断开连接',
        icon: 'success'
      });
    } catch (error) {
      wx.showModal({
        title: '错误',
        content: error.errMsg || '断开失败',
        showCancel: false
      });
    }
  },

  // 手动刷新数据
  async refreshData() {
    if (!this.bleManager.isConnected) {
      wx.showToast({
        title: '请先连接设备',
        icon: 'none'
      });
      return;
    }

    try {
      await this.bleManager.readTemp();
      await this.bleManager.readHumidity();
      wx.showToast({
        title: '刷新成功',
        icon: 'success'
      });
    } catch (error) {
      wx.showToast({
        title: '刷新失败',
        icon: 'none'
      });
    }
  },

  onUnload() {
    // 页面卸载时清理资源
    if (this.bleManager) {
      this.bleManager.destroy();
    }
  }
});
```

## 4. 其他服务使用示例

### 4.1 时间同步服务

```javascript
// 发送当前时间到设备
async syncTime() {
  const serviceId = '12340010-1234-5678-1234-56789abcdef0';
  const charId = '12340011-1234-5678-1234-56789abcdef0';
  
  // 获取当前时间戳（秒）
  const timestamp = Math.floor(Date.now() / 1000);
  
  // 转换为小端序字节数组
  const buffer = new ArrayBuffer(4);
  const view = new DataView(buffer);
  view.setUint32(0, timestamp, true); // true 表示小端序
  
  // 写入特征值
  wx.writeBLECharacteristicValue({
    deviceId: this.deviceId,
    serviceId: serviceId,
    characteristicId: charId,
    value: buffer,
    success: (res) => {
      console.log('时间同步成功', timestamp);
    },
    fail: (err) => {
      console.error('时间同步失败', err);
    }
  });
}
```

## 5. 注意事项

### 5.1 UUID 格式

- ✅ **必须使用小写字母**
- ✅ **必须使用完整 128 位格式**
- ❌ 不要使用 16/32 位 UUID 缩写（iOS 不支持）

### 5.2 数据格式

- 所有数据使用**小端序（Little Endian）**
- 温度：`int16_t`，单位 0.01°C
- 湿度：`uint16_t`，单位 0.01%

### 5.3 权限问题

- Android 需要定位权限（系统要求）
- iOS 需要在系统设置中允许小程序访问蓝牙
- 确保小程序已配置蓝牙相关权限

### 5.4 连接管理

- 页面卸载时必须清理 BLE 资源
- 监听连接状态变化，处理意外断开
- 建议添加重连机制

### 5.5 性能优化

- 合理控制通知频率（设备端已配置采集间隔）
- 避免频繁读写操作
- 及时停止不需要的扫描

## 6. 测试步骤

1. **配置小程序权限**：在 `app.json` 中添加蓝牙权限配置
2. **编译运行**：使用微信开发者工具编译并预览
3. **真机测试**：必须在真机上测试（模拟器不支持蓝牙）
4. **扫描设备**：确保设备已开机并广播
5. **连接测试**：验证连接、读取、通知功能

## 7. 常见问题

### Q: 扫描不到设备？
A: 检查：
- 设备是否已开机
- 设备是否在广播
- 小程序是否有蓝牙权限
- Android 是否有定位权限

### Q: 连接失败？
A: 检查：
- 设备是否被其他设备占用
- UUID 是否正确（大小写、格式）
- 设备是否支持连接

### Q: 读取数据失败？
A: 检查：
- 特征值属性是否支持 read
- 是否已正确初始化服务
- 数据格式解析是否正确

### Q: 通知收不到？
A: 检查：
- 是否已启用通知（notifyBLECharacteristicValueChange）
- 是否已监听数据变化（onBLECharacteristicValueChange）
- 设备端是否在发送通知
